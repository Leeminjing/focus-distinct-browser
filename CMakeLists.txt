cmake_minimum_required(VERSION 3.24)
project(distinct_browser VERSION 0.6.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Requires Windows and MSVC (x64).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WEBVIEW2_ROOT "${CMAKE_SOURCE_DIR}/.deps/webview2" CACHE PATH "Extracted Microsoft.Web.WebView2 SDK")
add_executable(distinct-browser WIN32 src/main.cpp src/host/host.cpp src/host/reconnect.cpp src/browser/pages.cpp src/browser/site_interactions.cpp src/window/native_toolbar.cpp)
target_include_directories(distinct-browser PRIVATE "${CMAKE_SOURCE_DIR}/src" "${WEBVIEW2_ROOT}/build/native/include")
target_compile_definitions(distinct-browser PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(distinct-browser PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_options(distinct-browser PRIVATE /MANIFEST:NO)
target_link_libraries(distinct-browser PRIVATE "${WEBVIEW2_ROOT}/build/native/x64/WebView2LoaderStatic.lib" user32 gdi32 ole32 oleaut32 shlwapi shell32 version windowsapp crypt32 bcrypt d2d1 dwrite dwmapi comctl32 advapi32)
target_sources(distinct-browser PRIVATE src/app.rc)

add_executable(lifecycle-tests tests/lifecycle_tests.cpp src/main.cpp src/host/host.cpp src/host/reconnect.cpp src/browser/pages.cpp src/browser/site_interactions.cpp src/window/native_toolbar.cpp src/app.rc)
target_include_directories(lifecycle-tests PRIVATE "${CMAKE_SOURCE_DIR}/src" "${WEBVIEW2_ROOT}/build/native/include")
target_compile_definitions(lifecycle-tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(lifecycle-tests PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_options(lifecycle-tests PRIVATE /MANIFEST:NO)
target_link_libraries(lifecycle-tests PRIVATE "${WEBVIEW2_ROOT}/build/native/x64/WebView2LoaderStatic.lib" user32 gdi32 ole32 oleaut32 shlwapi shell32 version windowsapp crypt32 bcrypt d2d1 dwrite dwmapi comctl32 advapi32)
enable_testing()
add_test(NAME lifecycle COMMAND lifecycle-tests)
add_executable(control-tests tests/control_tests.cpp)
target_compile_options(control-tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME control COMMAND control-tests)
add_executable(address-tests tests/address_tests.cpp)
target_compile_options(address-tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME address COMMAND address-tests)
add_executable(page-input-tests tests/page_input_tests.cpp)
target_compile_definitions(page-input-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(page-input-tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME page-input COMMAND page-input-tests)
add_executable(toolbar-component WIN32 tests/toolbar_component.cpp src/window/native_toolbar.cpp src/app.rc)
target_include_directories(toolbar-component PRIVATE "${CMAKE_SOURCE_DIR}/src")
target_compile_definitions(toolbar-component PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(toolbar-component PRIVATE /W4 /EHsc /utf-8)
target_link_options(toolbar-component PRIVATE /MANIFEST:NO)
target_link_libraries(toolbar-component PRIVATE user32 gdi32 ole32 d2d1 dwrite dwmapi comctl32 advapi32)

# Built for opt-in security acceptance only; never auto-create users in CTest.
add_executable(cross-user-worker tests/cross_user_worker.cpp)
add_executable(cross-user-runner tests/cross_user_runner.cpp)
foreach(target IN ITEMS cross-user-worker cross-user-runner)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  target_link_libraries(${target} PRIVATE advapi32)
endforeach()
target_link_libraries(cross-user-runner PRIVATE netapi32 bcrypt userenv)
